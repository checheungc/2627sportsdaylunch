系統雙重保險說明與 Google Apps Script (GAS) 後端設定
後端防競爭寫入 GAS 腳本 (已整合 LockService) : 
function doPost(e) {
  var lock = LockService.getScriptLock();
  
  try {
    lock.waitLock(10000); 
    
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    
    sheet.appendRow([
      new Date(),
      data.initial,
      data.day1Meal,
      data.day1Drink,
      data.day1Req || 'N/A',
      data.day2Meal,
      data.day2Drink,
      data.day2Req || 'N/A',
      data.totalEst
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({ 
      result: "success", 
      message: "Order recorded successfully" 
    })).setMimeType(ContentService.MimeType.JSON);

  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ 
      result: "error", 
      message: error.toString() 
    })).setMimeType(ContentService.MimeType.JSON);
    
  } finally {
    lock.releaseLock();
  }
}
